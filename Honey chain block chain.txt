<script>
(function(){
"use strict";
var box=document.getElementById("box");

function esc(s){
  return String(s==null?"":s).replace(/[&<>"']/g,function(c){
    return {"&":"&amp;","<":"&lt;",">":"&gt;","\"":"&quot;","'":"&#39;"}[c];
  });
}
function when(iso){
  if(!iso)return "—";
  var d=new Date(iso);
  return isNaN(d.getTime())?"—":d.toLocaleString("en-IN",{day:"2-digit",month:"short",year:"numeric"});
}
function kv(label,value){
  return "<div><span>"+label+"</span><b>"+(value==null||value===""?"—":value)+"</b></div>";
}

/* find batch code from ?code=, #code, or /verify/CODE */
function findCode(){
  var q="";
  try{ q=new URLSearchParams(location.search).get("code")||""; }catch(e){}
  if(q)return q.trim();
  var h=location.hash.replace(/^#\/?/,"");
  if(h)return h.trim();
  var m=location.pathname.match(/verify\/([^\/?#]+)/i);
  if(m)return decodeURIComponent(m[1]).trim();
  return "";
}
function clean(c){ return String(c||"").trim().toUpperCase().replace(/[^A-Z0-9_-]/g,""); }

function showInput(prefill){
  box.innerHTML='<div class="banner warn">Enter your jar code</div>'
    +'<form onsubmit="return HC.submit(this)">'
    +'<input id="hc-code" name="code" placeholder="e.g. HC001" value="'+esc(prefill||"")+'" autocomplete="off"/>'
    +'<button type="submit" style="width:100%">Verify</button></form>';
  var i=document.getElementById("hc-code");
  if(i)i.focus();
}
function loading(code){
  box.innerHTML='<div class="banner warn">⏳ Checking batch '+esc(code)+'…</div>'
    +'<p class="small muted">Contacting <code>/api/verify/'+esc(code)+'</code></p><div class="spin"></div>';
}
function showError(msg,code){
  box.innerHTML='<div class="banner err">⚠ Could not verify</div>'
    +'<p class="small">'+msg+'</p>'
    +'<div style="text-align:center;margin-top:12px">'
    +(code?'<button onclick="HC.go(\''+esc(code)+'\')">Retry</button> ':'')
    +'<button class="ghost" onclick="HC.showInput()">Enter another code</button></div>';
}

function go(rawCode){
  var code=clean(rawCode);
  if(!code){showInput();return;}
  if(location.protocol==="file:"){
    showError("You opened this file directly from disk. Open it through the server instead:<br><code>http://localhost:8000/verify/"+esc(code)+"</code>");
    return;
  }
  loading(code);
  try{history.replaceState(null,"","/verify/"+encodeURIComponent(code));}catch(e){}

  var xhr=new XMLHttpRequest();
  xhr.open("GET","/api/verify/"+encodeURIComponent(code),true);
  xhr.timeout=8000;
  xhr.onload=function(){handle(code,xhr);};
  xhr.onerror=function(){
    showError("Could not reach the server at <code>"+esc(location.host)+"</code>. "
      +"Is Honey Chain running?<br>Start it with <code>python main.py --seed --simulate</code>",code);
  };
  xhr.ontimeout=function(){
    showError("The server did not answer within 8 seconds. "
      +"Check the terminal running the backend for errors.",code);
  };
  try{xhr.send();}catch(e){showError("Request failed: "+esc(e.message),code);}
}

function handle(code,xhr){
  var body=xhr.responseText||"";
  var data=null;
  try{data=JSON.parse(body);}catch(e){}
  if(xhr.status===200&&data){render(data);return;}
  if(data&&data.detail){
    showError("HTTP "+xhr.status+": "+esc(data.detail),code);return;
  }
  showError("Server replied with HTTP "+xhr.status+" but not JSON — the backend probably crashed. "
    +"Check the uvicorn terminal.<br>Response was: <code>"+esc(body.slice(0,200))+"</code>",code);
}

function render(d){
  var b=d.batch||{},bk=d.beekeeper||{},hv=d.hive||{},q=d.quality,html="";
  html+=d.verified
    ?'<div class="banner ok">✓ Batch Verified — Ledger Intact</div>'
    :'<div class="banner warn">⏳ Details found — ledger confirmation pending</div>';
  html+='<div class="kv">';
  html+=kv("Batch ID",esc(b.code));
  html+=kv("Beekeeper",esc(bk.farm)+" ("+esc(bk.code)+")");
  html+=kv("Hive",esc(hv.code)+" — "+esc(hv.location));
  html+=kv("Collection",when(b.collection_date));
  html+=kv("Quantity",(b.qty!=null?esc(b.qty):"—")+" kg");
  html+=kv("Quality",q?esc(q.status):"Awaiting test");
  html+=kv("Grade",q?esc(q.grade):"—");
  html+=kv("Owner",esc(b.owner));
  html+=kv("Ledger",d.verified?"✓ Verified":"Pending");
  html+=kv("Block",b.block_index!=null?"#"+esc(b.block_index):"—");
  html+='</div><div class="wide"><span>Block hash</span><code>'+esc(b.block_hash||"—")+"</code></div>";
  if(q){
    html+='<p class="foot">Lab: '+esc(q.lab)+" • Moisture "+esc(q.moisture)+"% • Purity "+esc(q.purity)+"%</p>";
  }
  html+='<h3 class="sub">Journey of this honey</h3><ol class="tl">';
  var steps=d.trace||[];
  if(!steps.length)html+="<li>No journey events recorded yet.</li>";
  for(var i=0;i<steps.length;i++){
    var s=steps[i];
    var who=[s.actor,s.location].filter(function(x){return x;}).join(" • ");
    html+="<li><b>"+esc(s.title)+"</b><span class='t'>"+esc((s.when||"").slice(0,10))+"</span>"
      +"<div class='d'>"+esc(who)+" "+esc(s.details||"")+"</div>"
      +"<div class='tx'>"+(s.block?("⛓ ledger block #"+esc(s.block.index)):"⛓ pending")+"</div></li>";
  }
  html+="</ol><div style='text-align:center;margin-top:12px'>"
    +"<button class='ghost' onclick='HC.showInput()'>Verify another jar</button></div>";
  box.innerHTML=html;
}

window.HC={
  go:go,
  showInput:showInput,
  submit:function(f){go(f.code.value);return false;}
};

var initial=findCode();
if(initial){go(initial);}else{showInput();}
})();
</script>
